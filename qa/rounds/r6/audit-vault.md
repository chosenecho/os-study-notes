```python
import csv, hashlib, os, sys, time
from pathlib import Path

def sha256_file(p, chunk=4*1024*1024):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        while True:
            b = f.read(chunk)
            if not b: break
            h.update(b)
    return h.hexdigest()

root = Path(sys.argv[1]).resolve()
outdir = Path(sys.argv[2]); outdir.mkdir(parents=True, exist_ok=True)
files_rows, dirs_rows, errs = [], [], []
for st in sys.argv[3:]:
    base = root / st
    if not base.exists():
        errs.append(("MISSING_SUBTREE", st, "")); continue
    for dirpath, dirnames, filenames in os.walk(base, followlinks=False):
        dp = Path(dirpath); rel = dp.relative_to(root).as_posix()
        try:
            entries = list(os.scandir(dirpath))
        except OSError as e:
            errs.append(("SCANDIR", rel, repr(e))); entries = []
        dirs_rows.append({"dir": rel,
            "n_files_direct": sum(1 for e in entries if e.is_file(follow_symlinks=False)),
            "n_dirs_direct":  sum(1 for e in entries if e.is_dir(follow_symlinks=False))})
        for fn in filenames:
            p = dp / fn; rp = p.relative_to(root).as_posix()
            try:
                stt = p.stat(); size = stt.st_size; mtime = int(stt.st_mtime)
                if p.is_symlink(): digest = "SYMLINK"
                elif size == 0:    digest = hashlib.sha256(b"").hexdigest()
                else:              digest = sha256_file(p)
            except OSError as e:
                errs.append(("FILE", rp, repr(e))); continue
            ext = Path(fn).suffix.lower() or "(none)"
            files_rows.append({"path": rp, "dir": rel, "name": fn, "ext": ext,
                "size": size, "mtime": mtime, "sha256": digest,
                "subtree": st.split("/")[0], "depth": rp.count("/")})
with open(outdir/"files.csv","w",newline="",encoding="utf-8") as f:
    w=csv.DictWriter(f,fieldnames=["path","dir","name","ext","size","mtime","sha256","subtree","depth"])
    w.writeheader(); w.writerows(files_rows)
```
